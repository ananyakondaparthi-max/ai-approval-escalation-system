// ================= LOGIN & SESSION =================
let token = localStorage.getItem("token");

// Reads the user's id and role from the login token
function getUser() {
  if (!token) return null;
  try {
    const part = token.split(".")[1].replace(/-/g, "+").replace(/_/g, "/");
    const payload = JSON.parse(atob(part));
    if (payload.exp && payload.exp * 1000 < Date.now()) return null; // expired
    return payload; // { sub: user id, role: "employee" / "manager" / "admin" }
  } catch { return null; }
}

// Sends a request to the backend with the login token attached
async function api(path, method = "GET", body = null) {
  const res = await fetch(path, {
    method,
    headers: { "Content-Type": "application/json", "Authorization": "Bearer " + token },
    body: body ? JSON.stringify(body) : null,
  });
  if (res.status === 401) { logout(); throw new Error("Session expired. Please log in again."); }
  const data = await res.json().catch(() => ({}));
  if (!res.ok) throw new Error(errorText(data) || "Error " + res.status);
  return data;
}

function errorText(data) {
  if (typeof data.detail === "string") return data.detail;
  if (Array.isArray(data.detail)) return data.detail.map(d => d.loc.slice(-1)[0] + ": " + d.msg).join(", ");
  return "";
}

function esc(s) {
  return String(s ?? "").replace(/[&<>"']/g, c =>
    ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" }[c]));
}

document.getElementById("login-form").addEventListener("submit", async (e) => {
  e.preventDefault();
  const err = document.getElementById("login-error");
  err.textContent = "";
  const email = document.getElementById("login-email").value;
  const form = new URLSearchParams();
  form.append("username", email);
  form.append("password", document.getElementById("login-password").value);
  try {
    const res = await fetch("/auth/token", { method: "POST", body: form });
    const data = await res.json();
    if (!res.ok) throw new Error(errorText(data) || "Login failed");
    token = data.access_token;
    localStorage.setItem("token", token);
    localStorage.setItem("email", email);
    start();
  } catch (ex) { err.textContent = ex.message; }
});

  const ids = JSON.parse(localStorage.getItem(myPOKey()) || "[]");
  if (!ids.includes(id)) ids.unshift(id);
  localStorage.setItem(myPOKey(), JSON.stringify(ids));

async function loadPOs() {
  const list = document.getElementById("po-list");
  const user = getUser();
  list.innerHTML = '<p class="muted">Loading...</p>';
  let pos;
  try {
    pos = await api("/purchase-orders/");               // managers / admins
  } catch {
    const ids = JSON.parse(localStorage.getItem(myPOKey()) || "[]"); // employees
    pos = (await Promise.all(ids.map(id => api("/purchase-orders/" + id).catch(() => null))))
      .filter(Boolean);
  }
  if (!pos.length) { list.innerHTML = '<p class="muted">No purchase orders yet.</p>'; return; }

  list.innerHTML = pos.map(po => {
    const canSubmit = po.status === "DRAFT" && po.requester_id === user.sub;
    const canApprove = po.status === "PENDING_APPROVAL" &&
      (po.approver_id === user.sub || user.role === "admin");
    return `
      <div class="po-row">
        <div>
          <b>${esc(po.po_number)}</b> · ${esc(po.vendor_name)} · ₹${po.total_amount}
          <div class="muted">${esc(po.purpose)} · ${new Date(po.created_at).toLocaleDateString()}</div>
          ${po.rejection_reason ? `<div class="error">Reason: ${esc(po.rejection_reason)}</div>` : ""}
        </div>
        <div class="actions">
          <span class="badge ${po.status}">${po.status.replace(/_/g, " ")}</span>
          ${canSubmit ? `<button class="btn primary small" onclick="poAction('${po.id}','submit')">Submit</button>` : ""}
          ${canApprove ? `
            <button class="btn approve small" onclick="poAction('${po.id}','approve')">Approve</button>
            <button class="btn reject small" onclick="poAction('${po.id}','reject')">Reject</button>` : ""}
        </div>
      </div>`;
  }).join("");
}

async function poAction(id, action) {
  let body = {};
  if (action === "reject") {
    const reason = prompt("Reason for rejection:");
    if (!reason) return;
    body = { reason: reason, rejection_reason: reason };
  }
  try {
    await api(`/purchase-orders/${id}/${action}`, "POST", body);
    loadPOs();
  } catch (ex) { alert(ex.message); }
}

    if user_id == current_user.id:
        raise HTTPException(status_code=400, detail="You cannot deactivate your own account.")

    user.is_active = False
    db.commit()
    db.refresh(user)
    return user


@router.put("/users/{user_id}/manager", response_model=UserResponse)
def assign_manager(user_id: str, data: ManagerAssignment, db: Session = Depends(get_db),
                   current_user: User = Depends(admin_only)):
    """Set (or clear, with null) a user's manager."""
    user = _get_user(db, user_id)
    if data.manager_id:
        if data.manager_id == user.id:
            raise HTTPException(status_code=400, detail="A user cannot be their own manager.")
        manager = _get_user(db, data.manager_id)
        if manager.role not in MANAGER_ROLES:
            raise HTTPException(status_code=400, detail="That user is not a manager.")
    user.manager_id = data.manager_id
    db.commit()
    db.refresh(user)
    return user

def purchase_order_summary(db: Session = Depends(get_db), current_user: User = Depends(admin_only)):

    rows = db.query(PurchaseOrder.status, func.count(PurchaseOrder.id)).group_by(PurchaseOrder.status).all()
    for status_value, count in rows:
        counts[status_value] = count
    return {"total": sum(counts.values()), **{k.lower(): v for k, v in counts.items()}}

from datetime import datetime, timedelta

from fastapi import APIRouter, Depends, HTTPException
from fastapi.security import OAuth2PasswordRequestForm
from jose import jwt
from passlib.context import CryptContext
from sqlalchemy.orm import Session

from config import settings
from database import get_db
from models import RoleEnum, User
from schemas import LoginRequest, TokenResponse, UserCreate, UserResponse

router = APIRouter(prefix="/auth", tags=["auth"])
pwd_context = CryptContext(schemes=["bcrypt"], deprecated="auto")

VALID_ROLES = [role.value for role in RoleEnum]
MANAGER_ROLES = ("manager_l1", "manager_l2", "hr_admin")


def hash_password(password: str) -> str:
    return pwd_context.hash(password)


def verify_password(plain: str, hashed: str) -> bool:
    return pwd_context.verify(plain, hashed)


def create_token(data: dict) -> str:
    to_encode = data.copy()
    expire = datetime.utcnow() + timedelta(minutes=settings.ACCESS_TOKEN_EXPIRE_MINUTES)
    to_encode.update({"exp": expire})
    return jwt.encode(to_encode, settings.SECRET_KEY, algorithm=settings.ALGORITHM)


def authenticate(db: Session, email: str, password: str) -> User:
    user = db.query(User).filter(User.email == email).first()
    if not user or not verify_password(password, user.password):
        raise HTTPException(status_code=401, detail="Invalid credentials")
    if not user.is_active:
        raise HTTPException(status_code=403, detail="User account is inactive")
    return user


@router.post("/register", response_model=UserResponse)
def register(user: UserCreate, db: Session = Depends(get_db)):
    if user.role not in VALID_ROLES:
        raise HTTPException(status_code=400, detail=f"Role must be one of: {', '.join(VALID_ROLES)}")

    if db.query(User).filter(User.email == user.email).first():
        raise HTTPException(status_code=400, detail="Email already registered")

    if db.query(User).filter(User.employee_id == user.employee_id).first():
        raise HTTPException(status_code=400, detail="Employee ID already registered")

    if user.manager_id:
        manager = db.query(User).filter(User.id == user.manager_id).first()
        if not manager or manager.role not in MANAGER_ROLES:
            raise HTTPException(status_code=400, detail="manager_id must be the id of an existing manager")

    new_user = User(
        employee_id=user.employee_id,
        full_name=user.full_name,
        email=user.email,
        password=hash_password(user.password),
        role=user.role,
        department_id=user.department_id,
        manager_id=user.manager_id,
    )
    db.add(new_user)
    db.commit()
    db.refresh(new_user)
    return new_user


@router.post("/login", response_model=TokenResponse)
def login(request: LoginRequest, db: Session = Depends(get_db)):
    """JSON login used by the React frontend."""
    user = authenticate(db, request.email, request.password)
    token = create_token({"sub": str(user.id), "role": user.role})
    return {"access_token": token, "token_type": "bearer"}


@router.post("/token", response_model=TokenResponse)
def token(form_data: OAuth2PasswordRequestForm = Depends(), db: Session = Depends(get_db)):
    """Form login used by the Authorize button on /docs (username = email)."""
    user = authenticate(db, form_data.username, form_data.password)
    access_token = create_token({"sub": str(user.id), "role": user.role})
    return {"access_token": access_token, "token_type": "bearer"}

    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user)

        "sick": 10,
        "casual": 10,
        "annual": 20,
        "maternity": 90,
        "paternity": 15,
        "unpaid": 30
    }

    for leave_type, days in leave_types.items():

def calculate_working_days(start_date: datetime, end_date: datetime) -> float:
    days = 0
    current = start_date
    while current <= end_date:
        if current.weekday() < 5:  # Monday to Friday
            days += 1
        current += timedelta(days=1)
    return days


def find_approver(db: Session, user: User):

"""Common outflow services shared by all workflow types."""
import json
from datetime import datetime, timezone
from pathlib import Path
from typing import Any, Dict, Optional

from sqlalchemy.orm import Session

from models import AuditLog, DocumentRecord, NotificationLog
from tasks.notifications import send_email


def utcnow():
    return datetime.now(timezone.utc)


def log_audit(db: Session, *, workflow_id: str, workflow_type: str, event: str,
              actor_id: Optional[str] = None, details: Optional[Dict[str, Any]] = None,
              previous_status: Optional[str] = None, new_status: Optional[str] = None):
    row = AuditLog(workflow_id=workflow_id, workflow_type=workflow_type, event=event,
                   actor_id=actor_id, previous_status=previous_status,
                   new_status=new_status, details=json.dumps(details or {}, default=str),
                   created_at=utcnow())
    db.add(row)
    return row


def queue_notification(db: Session, *, workflow_id: str, workflow_type: str,
                       event: str, to_email: str, subject: str, body: str):
    row = NotificationLog(workflow_id=workflow_id, workflow_type=workflow_type,
                          event=event, recipient=to_email, subject=subject,
                          body=body, status="queued", created_at=utcnow())
    db.add(row)
    db.flush()
    row.status = send_email(to_email, subject, body)
    return row


def generate_completion_document(db: Session, *, workflow_id: str, workflow_type: str,
                                 event: str, data: Dict[str, Any]):
    from reportlab.lib.pagesizes import A4
    from reportlab.pdfgen import canvas

    out_dir = Path("generated_documents")
    out_dir.mkdir(exist_ok=True)
    safe_id = "".join(c for c in workflow_id if c.isalnum() or c in "-_") or "workflow"
    filename = f"{safe_id}_{event.lower()}_report.pdf"
    path = out_dir / filename

    c = canvas.Canvas(str(path), pagesize=A4)
    width, height = A4
    y = height - 60
    c.setFont("Helvetica-Bold", 18)
    c.drawString(50, y, "Workflow Completion Report")
    y -= 35
    c.setFont("Helvetica", 10)
    c.drawString(50, y, f"Workflow ID: {workflow_id}")
    y -= 18
    c.drawString(50, y, f"Workflow Type: {workflow_type}")
    y -= 18
    c.drawString(50, y, f"Event: {event}")
    y -= 18
    c.drawString(50, y, f"Generated: {utcnow().isoformat()}")
    y -= 30
    c.setFont("Helvetica-Bold", 12)
    c.drawString(50, y, "Workflow Data")
    y -= 20
    c.setFont("Helvetica", 10)
    for key, value in data.items():
        text = f"{key}: {value}"
        for chunk in [text[i:i + 95] for i in range(0, len(text), 95)]:
            if y < 55:
                c.showPage()
                y = height - 55
                c.setFont("Helvetica", 10)
            c.drawString(55, y, chunk)
            y -= 16
    c.save()

    row = DocumentRecord(workflow_id=workflow_id, workflow_type=workflow_type,
                         event=event, filename=filename, path=str(path),
                         created_at=utcnow())
    db.add(row)
    return row


def process_event(db: Session, *, workflow_id: str, workflow_type: str, event: str,
                  actor_id: Optional[str] = None, recipient_email: Optional[str] = None,
                  notification_subject: Optional[str] = None,
                  notification_body: Optional[str] = None,
                  data: Optional[Dict[str, Any]] = None,
                  previous_status: Optional[str] = None,
                  new_status: Optional[str] = None):
    data = data or {}
    audit = log_audit(db, workflow_id=workflow_id, workflow_type=workflow_type,
                      event=event, actor_id=actor_id, details=data,
                      previous_status=previous_status, new_status=new_status)

    notification = None
    if recipient_email and notification_subject and notification_body:
        notification = queue_notification(db, workflow_id=workflow_id,
                                          workflow_type=workflow_type, event=event,
                                          to_email=recipient_email,
                                          subject=notification_subject,
                                          body=notification_body)

    document = None
    if event.upper() == "COMPLETED":
        document = generate_completion_document(db, workflow_id=workflow_id,
                                                workflow_type=workflow_type,
                                                event=event, data=data)
    return audit, notification, document

"""Purchase order logic and approval workflow."""
from datetime import datetime, timezone
from uuid import uuid4

from sqlalchemy.orm import Session

from models import User
from po_models import PurchaseOrder, PurchaseOrderItem
from services.outflow_service import process_event

WORKFLOW_TYPE = "purchase_order"
APPROVER_ROLES = ("manager_l1", "manager_l2", "hr_admin")
VIEW_ALL_ROLES = ("manager_l2", "hr_admin")
EDITABLE_STATUSES = ("DRAFT", "REJECTED")
CANCELLABLE_STATUSES = ("DRAFT", "PENDING_APPROVAL")


def utcnow():
    return datetime.now(timezone.utc)


def generate_po_number() -> str:
    return f"PO-{datetime.now().year}-{uuid4().hex[:8].upper()}"


def _build_items(items_data):
    items, total = [], 0.0
    for d in items_data:
        line_total = round(d.quantity * d.unit_price, 2)
        items.append(PurchaseOrderItem(
            item_name=d.item_name,
            item_description=d.item_description,
            quantity=d.quantity,
            unit_price=d.unit_price,
            total_price=line_total,
        ))
        total += line_total
    return items, round(total, 2)


def find_approver(db: Session, user: User):
    """The user's own manager, otherwise the first available approver."""
    if user.manager_id:
        manager = db.query(User).filter(User.id == user.manager_id, User.is_active == True).first()
        if manager and manager.role in APPROVER_ROLES and manager.id != user.id:
            return manager
    for role in APPROVER_ROLES:
        approver = db.query(User).filter(
            User.role == role, User.is_active == True, User.id != user.id
        ).first()
        if approver:
            return approver
    return None


def can_view(po: PurchaseOrder, user: User) -> bool:
    return (
        po.requester_id == user.id
        or po.approver_id == user.id
        or user.role in VIEW_ALL_ROLES
    )


def can_decide(po: PurchaseOrder, user: User) -> bool:
    if po.requester_id == user.id:
        return False
    if user.role in VIEW_ALL_ROLES:
        return True
    return user.role in APPROVER_ROLES and po.approver_id == user.id


def _po_html(po: PurchaseOrder, heading: str) -> str:
    return (
        f"<p>{heading}</p>"
        f"<p><b>PO Number:</b> {po.po_number}</p>"
        f"<p><b>Department:</b> {po.department}</p>"
        f"<p><b>Vendor:</b> {po.vendor_name}</p>"
        f"<p><b>Purpose:</b> {po.purpose}</p>"
        f"<p><b>Total:</b> {po.total_amount}</p>"
    )


def _record(db: Session, po: PurchaseOrder, event: str, actor_id: str,
            old_status, new_status, recipient=None, subject=None, body=None, extra=None):
    data = {
        "po_number": po.po_number,
        "department": po.department,
        "vendor_name": po.vendor_name,
        "purpose": po.purpose,
        "total_amount": po.total_amount,
        "items": [f"{i.item_name} x{i.quantity} @ {i.unit_price}" for i in po.items],
    }
    if extra:
        data.update(extra)
    process_event(
        db,
        workflow_id=str(po.id),
        workflow_type=WORKFLOW_TYPE,
        event=event,
        actor_id=actor_id,
        recipient_email=recipient,
        notification_subject=subject if recipient else None,
        notification_body=body if recipient else None,
        data=data,
        previous_status=old_status,
        new_status=new_status,
    )

def update_purchase_order(db: Session, po: PurchaseOrder, user: User, data):
    if po.status not in EDITABLE_STATUSES:
        raise ValueError("Only DRAFT or REJECTED Purchase Orders can be edited.")
    if po.requester_id != user.id and user.role != "hr_admin":
        raise PermissionError("You can only edit your own Purchase Order.")

    for field in ("department", "vendor_name", "vendor_email", "vendor_phone", "purpose", "description"):
        value = getattr(data, field)
        if value is not None:
            setattr(po, field, value)

    if data.items is not None:
        if not data.items:
            raise ValueError("A Purchase Order needs at least one item.")
        items, total = _build_items(data.items)
        po.items.clear()
        po.items.extend(items)
        po.total_amount = total

    po.updated_at = utcnow()
    db.commit()
    db.refresh(po)
    return po

def submit_purchase_order(db: Session, po: PurchaseOrder, user: User):
    if po.status != "DRAFT":
        raise ValueError("Only DRAFT Purchase Orders can be submitted.")
    if po.requester_id != user.id:
        raise PermissionError("You can only submit your own Purchase Order.")
    if not po.items or po.total_amount <= 0:
        raise ValueError("Purchase Order must have at least one item and a total above zero.")

    approver = find_approver(db, user)
    if not approver:
        raise ValueError("No approver available. Register a manager_l1 user first.")

    old = po.status
    po.status = "PENDING_APPROVAL"
    po.approver_id = approver.id
    po.submitted_at = utcnow()
    po.updated_at = utcnow()

    _record(db, po, "SUBMITTED", user.id, old, po.status,
            recipient=approver.email,
            subject=f"Purchase Order {po.po_number} needs your approval",
            body=_po_html(po, f"A Purchase Order from {user.full_name} needs your approval."))
    db.commit()
    db.refresh(po)
    return po


def approve_purchase_order(db: Session, po: PurchaseOrder, user: User, remarks=None):
    if po.status != "PENDING_APPROVAL":
        raise ValueError("Only Purchase Orders pending approval can be approved.")
    if not can_decide(po, user):
        raise PermissionError("You are not allowed to approve this Purchase Order.")

    old = po.status
    po.status = "APPROVED"
    po.approver_id = user.id
    po.approved_at = utcnow()
    po.updated_at = utcnow()

    requester = db.query(User).filter(User.id == po.requester_id).first()
    _record(db, po, "APPROVED", user.id, old, po.status,
            recipient=requester.email if requester else None,
            subject=f"Your Purchase Order {po.po_number} has been approved",
            body=_po_html(po, f"Your Purchase Order was approved by {user.full_name}.")
                 + f"<p><b>Remarks:</b> {remarks or 'None'}</p>",
            extra={"remarks": remarks})

    # COMPLETED makes the shared document generator create a PDF.
    _record(db, po, "COMPLETED", user.id, po.status, po.status,
            extra={"approved_by": user.full_name, "remarks": remarks})

    db.commit()
    db.refresh(po)
    return po


def reject_purchase_order(db: Session, po: PurchaseOrder, user: User, reason: str):
    if po.status != "PENDING_APPROVAL":
        raise ValueError("Only Purchase Orders pending approval can be rejected.")
    if not can_decide(po, user):
        raise PermissionError("You are not allowed to reject this Purchase Order.")
    if not reason or not reason.strip():
        raise ValueError("A rejection reason is required.")

    old = po.status
    po.status = "REJECTED"
    po.approver_id = user.id
    po.rejection_reason = reason.strip()
    po.rejected_at = utcnow()
    po.updated_at = utcnow()

    requester = db.query(User).filter(User.id == po.requester_id).first()

def resubmit_purchase_order(db: Session, po: PurchaseOrder, user: User):
    if po.status != "REJECTED":
        raise ValueError("Only REJECTED Purchase Orders can be resubmitted.")
    if po.requester_id != user.id:
        raise PermissionError("You can only resubmit your own Purchase Order.")

    approver = find_approver(db, user)
    if not approver:
        raise ValueError("No approver available.")

    old = po.status
    po.status = "PENDING_APPROVAL"
    po.approver_id = approver.id
    po.submitted_at = utcnow()
    po.rejected_at = None
    po.rejection_reason = None
    po.updated_at = utcnow()

    _record(db, po, "RESUBMITTED", user.id, old, po.status,
            recipient=approver.email,
            subject=f"Purchase Order {po.po_number} was resubmitted",
            body=_po_html(po, f"{user.full_name} corrected and resubmitted this Purchase Order."))
    db.commit()
    db.refresh(po)
    return po


def cancel_purchase_order(db: Session, po: PurchaseOrder, user: User):
    if po.requester_id != user.id and user.role != "hr_admin":
        raise PermissionError("You can only cancel your own Purchase Order.")
    if po.status not in CANCELLABLE_STATUSES:
        raise ValueError("This Purchase Order cannot be cancelled.")

    old = po.status
    po.status = "CANCELLED"
    po.updated_at = utcnow()
    _record(db, po, "CANCELLED", user.id, old, po.status)
    db.commit()
    db.refresh(po)
    return po

from datetime import datetime, timezone

from celery_app import celery_app
from database import SessionLocal
from models import ApprovalHistory, LeaveRequest, StatusEnum, User
from tasks.notifications import send_email


@celery_app.task
def check_and_escalate():
    db = SessionLocal()
    try:
        open_requests = db.query(LeaveRequest).filter(
            LeaveRequest.status.in_(["pending", "escalated"])
        ).all()

        for request in open_requests:
            hours_pending = (
                datetime.now(timezone.utc) - request.created_at.replace(tzinfo=timezone.utc)
            ).total_seconds() / 3600

            if hours_pending >= 2/60 and request.escalation_level == "manager_l1":
                escalate_request.delay(str(request.id), "manager_l2")
            elif hours_pending >=4/60 and request.escalation_level == "manager_l2":
                escalate_request.delay(str(request.id), "hr_admin")
    finally:
        db.close()


@celery_app.task
def escalate_request(request_id: str, escalate_to: str):
    db = SessionLocal()
    try:
        request = db.query(LeaveRequest).filter(LeaveRequest.id == request_id).first()
        if not request:
            return {"error": "Request not found"}

        new_approver = db.query(User).filter(
            User.role == escalate_to,
            User.is_active == True,
        ).first()
        if not new_approver:
            return {"error": f"No approver found for role {escalate_to}"}

        request.escalation_level = escalate_to
        request.current_approver_id = new_approver.id
        request.status = StatusEnum.escalated
        request.escalated_at = datetime.now(timezone.utc)

        db.add(ApprovalHistory(
            request_id=request.id,
            actor_id=new_approver.id,
            action="escalated",
            comments=f"Auto-escalated to {escalate_to}",
        ))
        db.commit()

        send_email(
            new_approver.email,
            f"Escalated Leave Request: {request.leave_type}",
            f"<p>A request has been escalated to you.</p>"
            f"<p><b>Leave Type:</b> {request.leave_type}</p>"
            f"<p><b>Reason:</b> {request.reason}</p>",
        )
        return {"status": "escalated", "request_id": request_id}
    finally:
        db.close()

import smtplib
from email.mime.multipart import MIMEMultipart
from email.mime.text import MIMEText

from celery_app import celery_app
from config import settings


@celery_app.task
def send_email_notification(to_email: str, subject: str, body: str):
    try:
        msg = MIMEMultipart()
        msg["From"] = settings.SMTP_USER
        msg["To"] = to_email
        msg["Subject"] = subject
        msg.attach(MIMEText(body, "html"))

        if settings.SMTP_USER and settings.SMTP_PASSWORD:
            server = smtplib.SMTP(settings.SMTP_HOST, settings.SMTP_PORT)
            server.starttls()
            server.login(settings.SMTP_USER, settings.SMTP_PASSWORD)
            server.send_message(msg)
            server.quit()
        else:
            print(f"[EMAIL MOCK] To: {to_email} | Subject: {subject}")
            print(f"[EMAIL MOCK] Body: {body}")

        return {"status": "sent", "to": to_email}
    except Exception as e:
        print(f"Email error: {e}")
        return {"status": "failed", "error": str(e)}


def send_email(to_email: str, subject: str, body: str) -> str:
    """Use Celery if it's turned on, otherwise send right away.

    Without SMTP settings, the email is printed in the terminal as [EMAIL MOCK].
    """
    if settings.USE_CELERY:
        try:
            send_email_notification.delay(to_email, subject, body)
            return "queued"
        except Exception as exc:
            print(f"Celery/Redis not available, sending directly: {exc}")

    result = send_email_notification(to_email, subject, body)
    return result.get("status", "sent")
